cmake_minimum_required(VERSION 3.22)
project(juego CXX C)
set(CMAKE_CXX_STANDARD 17)
set(GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
# DOS librerias en el mismo APK: SUB 2 (libjuego.so, la que abre Android) y SUB 1 (libjuego_sub1.so, la abre la primera segun sub.txt)
foreach(NOM juego juego_sub1)
  if(NOM STREQUAL "juego")
    set(SV 2)
  else()
    set(SV 1)
  endif()
  add_library(${NOM} SHARED ../../../../../src/main.cpp ../../../../../src/red_sock.cpp ${GLUE}/android_native_app_glue.c)
  target_include_directories(${NOM} PRIVATE ${GLUE} ../../../../../src)
  target_compile_definitions(${NOM} PRIVATE MOVIL ANDROID_GAME SUB_VAL=${SV})
  target_compile_options(${NOM} PRIVATE -O2 -ffast-math)
  target_link_libraries(${NOM} android log EGL GLESv3 atomic)
  set_target_properties(${NOM} PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate")
endforeach()
