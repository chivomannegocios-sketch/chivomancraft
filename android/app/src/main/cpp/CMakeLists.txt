cmake_minimum_required(VERSION 3.22)
project(juego CXX C)
set(CMAKE_CXX_STANDARD 17)
set(GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(juego SHARED ../../../../../src/main.cpp ../../../../../src/red_sock.cpp ${GLUE}/android_native_app_glue.c)
target_include_directories(juego PRIVATE ${GLUE} ../../../../../src)
target_compile_definitions(juego PRIVATE MOVIL ANDROID_GAME)
target_compile_options(juego PRIVATE -O2 -ffast-math)
target_link_libraries(juego android log EGL GLESv3 atomic)
set_target_properties(juego PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate")
